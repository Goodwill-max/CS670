cmake_minimum_required(VERSION 3.18)
project(contact_discovery LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(FETCH_AUTO ON CACHE BOOL "Fetch upstream dependencies")
set(ENABLE_DRRN_PSI ON CACHE BOOL "BGI DPF")
set(ENABLE_KKRT_PSI ON CACHE BOOL "KKRT PSI")
set(ENABLE_RELIC ON CACHE BOOL "Base OTs")
set(LIBPSI_BUILD_FRONTEND OFF CACHE BOOL "Exclude upstream standalone benchmarks" FORCE)
if(NOT MSVC)
  add_compile_options(-maes -msse4.1 -mpclmul)
endif()
add_subdirectory(vendor/libPSI)
add_executable(contact_discovery src/main.cpp)
target_link_libraries(contact_discovery PRIVATE libPSI)
if(NOT MSVC)
  target_compile_options(contact_discovery PRIVATE -maes -msse4.1 -mpclmul)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:contact_discovery>)
set_tests_properties(protocol PROPERTIES TIMEOUT 180)
add_test(NAME primitives COMMAND contact_discovery selftest)
