cmake_minimum_required(VERSION 3.18)
project(psi_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ORIGINAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
# Reuse the already installed cryptographic backend, not the original executable.
set(LIBPSI_THIRDPARTY_DIR "${ORIGINAL_ROOT}/vendor/libPSI/out/install/linux" CACHE PATH "Backend installation")
set(FETCH_AUTO OFF CACHE BOOL "Use existing backend" FORCE)
set(ENABLE_DRRN_PSI ON CACHE BOOL "BGI DPF" FORCE)
set(ENABLE_KKRT_PSI ON CACHE BOOL "KKRT PSI" FORCE)
set(ENABLE_RELIC ON CACHE BOOL "Base OT backend" FORCE)
set(LIBPSI_BUILD_FRONTEND OFF CACHE BOOL "Exclude benchmark" FORCE)
if(NOT MSVC)
  add_compile_options(-maes -msse4.1 -mpclmul)
endif()
add_subdirectory("${ORIGINAL_ROOT}/vendor/libPSI" backend)
add_executable(kkrt_psi src/psi_main.cpp)
target_link_libraries(kkrt_psi PRIVATE libPSI)
add_executable(pir_psi_measured src/pir_psi_measured.cpp)
target_link_libraries(pir_psi_measured PRIVATE libPSI)
